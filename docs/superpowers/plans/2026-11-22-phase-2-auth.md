# Phase 2 Auth Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace demo access (Basic-auth password + "View as" cookie) with Microsoft 365 sign-in via Supabase Auth, role from `employees.role`, data access enforced by RLS.

**Architecture:** `proxy.ts` refreshes the Supabase session cookie and gates every route except `/login` and `/auth/callback`. `lib/supabase.ts` `db()` becomes a per-request `@supabase/ssr` client using the anon key + user cookies, so every query in `lib/data.ts` runs as the user under RLS. Matching a login to an employee happens only in SQL (`current_employee_id()`); `lib/role.ts` calls it via RPC.

**Tech Stack:** Next.js 16.4 App Router (`proxy.ts`, not middleware), `@supabase/supabase-js` 2.x, `@supabase/ssr` 0.12.x, Postgres RLS.

Spec: [docs/superpowers/specs/2026-11-22-phase-2-auth-design.md](../specs/2026-11-22-phase-2-auth-design.md)

## Global Constraints

- Next 16: read `node_modules/next/dist/docs/01-app/03-api-reference/03-file-conventions/proxy.md` before touching `proxy.ts`. `cookies()` / `headers()` are async.
- Only new dependency: `@supabase/ssr`.
- Browser never talks to Supabase. Env vars stay server-only (no `NEXT_PUBLIC_`).
- App env: `SUPABASE_URL`, `SUPABASE_ANON_KEY`. Seed env: `SUPABASE_URL`, `SUPABASE_SERVICE_ROLE_KEY`, optional `SEED_ADMIN_EMAIL`. `DEMO_PASSWORD` removed.
- Active employee = `exit_date is null or exit_date >= current_date`. Email match is case-insensitive.
- No insert/update/delete policies in this phase.
- `lib/data.ts` exported function signatures do not change.
- Work on branch `phase-2-auth`. `npm test`, `npm run lint`, `npm run build` pass at the end of every task.

```sql
-- RLS check. Paste into Supabase SQL editor after seeding + migration 0003.
-- Impersonates users inside a transaction and rolls back. Any failed assert aborts with its message.
begin;

-- 1. Plain employee with a manager
select set_config('request.jwt.claims', json_build_object('email', (
  select upper(email) from employees  -- upper(): proves case-insensitive match
  where role = 'employee' and exit_date is null and manager_id is not null limit 1))::text, true);
set local role authenticated;
do $$ begin
  assert current_employee_id() is not null, 'employee matched by email';
  assert (select count(*) from employees) > 1, 'employee can read directory';
  assert (select count(distinct employee_id) from leave_balances) <= 1, 'employee sees only own balances';
  assert not exists (select 1 from leave_balances where employee_id <> current_employee_id()), 'no other balances';
  assert not exists (select 1 from absences where status = 'pending' and employee_id <> current_employee_id()), 'no others pending leave';
end $$;
reset role;

-- 2. Unknown account
select set_config('request.jwt.claims', '{"email":"nobody@example.com"}', true);
set local role authenticated;
do $$ begin
  assert current_employee_id() is null, 'unknown email not matched';
  assert (select count(*) from employees) = 0, 'unknown user reads no employees';
  assert (select count(*) from absences) = 0, 'unknown user reads no absences';
end $$;
reset role;

-- 3. Leaver (exit date in the past) gets nothing
select set_config('request.jwt.claims', json_build_object('email', (
  select email from employees where exit_date < current_date limit 1))::text, true);
set local role authenticated;
do $$ begin
  assert current_employee_id() is null, 'leaver not matched';
end $$;
reset role;

-- 4. HR admin sees everyone's balances and pending leave
select set_config('request.jwt.claims', json_build_object('email', (
  select email from employees where role = 'hr_admin' and exit_date is null limit 1))::text, true);
set local role authenticated;
do $$ begin
  assert is_hr(), 'hr_admin is HR';
  assert (select count(distinct employee_id) from leave_balances) > 1, 'HR sees all balances';
end $$;
reset role;

rollback;
select 'RLS check passed' as result;
```

Note: if the seed has no leaver, case 3's subquery is null, the claim email is null, and the assert still passes trivially — acceptable; the seed does create leavers (`exit_date` set).

- [ ] **Step 3: Write the migration** — `supabase/migrations/0003_auth_rls.sql`

```sql
-- Phase 2: signed-in users read data through RLS. Matching a login to an employee lives only here.
-- security definer so policies can look up employees without recursing into employees' own RLS.

create or replace function public.current_employee_id() returns uuid
language sql stable security definer set search_path = public as $$
  select id from employees
  where lower(email) = lower(auth.jwt() ->> 'email')
    and (exit_date is null or exit_date >= current_date)
  limit 1
$$;

create or replace function public.is_hr() returns boolean
language sql stable security definer set search_path = public as $$
  select exists (select 1 from employees
                 where id = current_employee_id() and role in ('hr_admin', 'super_admin'))
$$;

create or replace function public.manages(emp uuid) returns boolean
language sql stable security definer set search_path = public as $$
  select exists (select 1 from employees where id = emp and manager_id = current_employee_id())
$$;

revoke execute on function public.current_employee_id(), public.is_hr(), public.manages(uuid) from public, anon;
grant execute on function public.current_employee_id(), public.is_hr(), public.manages(uuid) to authenticated;

grant usage on schema public to authenticated;
grant select on all tables in schema public to authenticated;

-- Company-wide reference data and directory: any active employee.
do $$
declare t text;
begin
  foreach t in array array['offices','brands','departments','employees','sops','events','announcements','job_openings'] loop
    execute format('create policy "active staff read" on %I for select to authenticated using (current_employee_id() is not null)', t);
  end loop;
end $$;

-- Who's out is public to staff; pending requests are private to self, manager, HR.
create policy "staff read absences" on absences for select to authenticated using (
  current_employee_id() is not null
  and (status = 'approved' or employee_id = current_employee_id() or manages(employee_id) or is_hr())
);

create policy "own team or HR read balances" on leave_balances for select to authenticated using (
  employee_id = current_employee_id() or manages(employee_id) or is_hr()
);
```

```ts
import "server-only";
import { cache } from "react";
import { cookies } from "next/headers";
import { createServerClient } from "@supabase/ssr";

// Per-request client bound to the signed-in user's session cookies (anon key), so every query runs under RLS.
// The service-role key is used only by scripts/seed.ts.
export const db = cache(() => {
  const url = process.env.SUPABASE_URL;
  const key = process.env.SUPABASE_ANON_KEY;
  if (!url || !key) throw new Error("SUPABASE_URL / SUPABASE_ANON_KEY are not set");
  return createServerClient(url, key, {
    cookies: {
      getAll: async () => (await cookies()).getAll(),
      setAll: async (list) => {
        try {
          const jar = await cookies();
          list.forEach(({ name, value, options }) => jar.set(name, value, options));
        } catch {
          // Server Components can't set cookies; proxy.ts refreshes the session on every request.
        }
      },
    },
  });
});
```

- [ ] **Step 3: Rewrite `proxy.ts`**

```ts
import { NextResponse, type NextRequest } from "next/server";
import { createServerClient } from "@supabase/ssr";

const PUBLIC = ["/login", "/auth/callback"];

// Refreshes the Supabase session cookie and sends signed-out visitors to /login.
// Pages re-check identity server-side (lib/role.ts); this is the first gate, not the only one.
export async function proxy(req: NextRequest) {
  let res = NextResponse.next({ request: req });
  const supabase = createServerClient(process.env.SUPABASE_URL!, process.env.SUPABASE_ANON_KEY!, {
    cookies: {
      getAll: () => req.cookies.getAll(),
      setAll: (list) => {
        list.forEach(({ name, value }) => req.cookies.set(name, value));
        res = NextResponse.next({ request: req });
        list.forEach(({ name, value, options }) => res.cookies.set(name, value, options));
      },
    },
  });
  const { data: { user } } = await supabase.auth.getUser();
  if (!user && !PUBLIC.includes(req.nextUrl.pathname)) {
    return NextResponse.redirect(new URL("/login", req.url));
  }
  return res;
}

export const config = { matcher: ["/((?!_next/static|_next/image|favicon.ico|logo.svg).*)"] };
```

```ts
import "server-only";
import { cache } from "react";
import { redirect } from "next/navigation";
import { db } from "./supabase";
import type { Role } from "./constants";

export type Me = { id: string; full_name: string; title: string; office_id: string; department_id: string; role: Role };

/** The signed-in, active employee. Matching rules live in SQL (current_employee_id). Unmatched → /no-access. */
export const currentEmployee = cache(async (): Promise<Me> => {
  const { data: id } = await db().rpc("current_employee_id");
  if (!id) redirect("/no-access");
  const { data, error } = await db()
    .from("employees")
    .select("id, full_name, title, office_id, department_id, role")
    .eq("id", id)
    .single();
  if (error || !data) redirect("/no-access");
  return data as Me;
});

export const currentRole = async (): Promise<Role> => (await currentEmployee()).role;

export const canViewAll = (r: Role) => r !== "employee";
export const canEditSops = (r: Role) => r === "hr_admin" || r === "super_admin";
```
